def evaluate(value: int):
    let chained = value.strip().length()
    let powered = -2 ** 3
    let quotient = 7 // 2
    let bits = 5 & 3 | 1 ^ 2
    let shifted = 1 << 4
    let unary = +~value
    print(chained + powered + quotient + bits + shifted + unary)

def classify(value: int) -> int:
    if value > 10:
        return value
    elif value == 10:
        return 0
    else:
        while value < 10:
            let value = value + 1
        return value
